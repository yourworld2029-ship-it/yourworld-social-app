import { useState, useRef } from 'react';
import { createFileRoute, useParams, useNavigate } from '@tanstack/react-router';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { supabase } from '../lib/supabase';
import { ArrowLeft, Share2, Send, AlertCircle } from 'lucide-react';
import { formatDistanceToNow } from 'date-fns';
import { Avatar, AvatarFallback, AvatarImage } from '../components/ui/avatar';
import { Button } from '../components/ui/button';
import { Input } from '../components/ui/input';
import { useAuth } from '../contexts/AuthContext';
import { toast } from 'sonner';

export const Route = createFileRoute('/video/$videoId')({
  component: VideoWatchPage,
  errorComponent: () => <VideoErrorFallback/>,
  notFoundComponent: () => <VideoErrorFallback/>,
});

function VideoErrorFallback() {
  const navigate = useNavigate();
  return (
    <div className="min-h-screen bg-black text-white flex flex-col items-center justify-center p-6 gap-4 text-center">
      <AlertCircle className="w-12 h-12 text-pink-500"/>
      <h2 className="text-xl font-bold">Video not available</h2>
      <p className="text-gray-400 text-sm max-w-xs">This video could not be loaded or was removed.</p>
      <Button onClick="{()"> window.history.length > 1 ? window.history.back() : navigate({ to: '/' })}
        className="bg-pink-600 hover:bg-pink-700 text-white px-6 py-2 rounded-full mt-2"
      >
        <ArrowLeft className="w-4 h-4 mr-2"/> Go Back
      </Button>
    </div>
  );
}

export default function VideoWatchPage() {
  const params = useParams({ strict: false }) as { videoId?: string };
  const rawId = params?.videoId || '';
  const cleanId = rawId.replace(/[^a-zA-Z0-9-]/g, '');
  const navigate = useNavigate();
  const { user } = useAuth();
  const queryClient = useQueryClient();

  const [commentText, setCommentText] = useState('');
  const videoRef = useRef<HTMLVideoElement>(null);

  const { data: video, isLoading, isError } = useQuery({
    queryKey: ['video-detail', cleanId],
    queryFn: async () => {
      if (!cleanId) return null;
      const { data, error } = await supabase
        .from('posts')
        .select(`
          *,
          user:profiles!user_id (
            id,
            username,
            full_name,
            avatar_url
          )
        `)
        .eq('id', cleanId)
        .maybeSingle();

      if (error) {
        console.error('Error fetching video:', error);
        return null;
      }
      return data;
    },
    enabled: !!cleanId,
    retry: 1,
  });

  const { data: comments = [] } = useQuery({
    queryKey: ['video-comments', cleanId],
    queryFn: async () => {
      if (!cleanId) return [];
      const { data, error } = await supabase
        .from('comments')
        .select(`
          *,
          user:profiles!user_id (
            id,
            username,
            avatar_url
          )
        `)
        .eq('post_id', cleanId)
        .order('created_at', { ascending: false });

      if (error) return [];
      return data || [];
    },
    enabled: !!cleanId,
  });

  const addCommentMutation = useMutation({
    mutationFn: async (text: string) => {
      if (!user) throw new Error('Must be logged in');
      const { data, error } = await supabase
        .from('comments')
        .insert({
          post_id: cleanId,
          user_id: user.id,
          content: text.trim(),
        })
        .select()
        .single();

      if (error) throw error;
      return data;
    },
    onSuccess: () => {
      setCommentText('');
      queryClient.invalidateQueries({ queryKey: ['video-comments', cleanId] });
      toast.success('Comment added');
    },
    onError: () => {
      toast.error('Failed to post comment');
    },
  });

  const handleShare = async () => {
    if (navigator.share) {
      try {
        await navigator.share({
          title: video?.title || 'Watch Video',
          url: window.location.href,
        });
      } catch (err) {}
    } else {
      navigator.clipboard.writeText(window.location.href);
      toast.success('Link copied to clipboard');
    }
  };

  if (isLoading) {
    return (
      <div className="min-h-screen bg-black text-white flex flex-col items-center justify-center">
        <div className="w-10 h-10 border-4 border-pink-500 border-t-transparent rounded-full animate-spin mb-4" />
        <p className="text-gray-400 text-sm">Loading video...</p>
      </div>
    );
  }

  if (isError || !video) {
    return <VideoErrorFallback/>;
  }

  const mediaUrl = video.media_url || video.video_url || video.url || '';
  const timeAgo = video.created_at ? formatDistanceToNow(new Date(video.created_at), { addSuffix: true }) : '';

  return (
    <div className="min-h-screen bg-black text-white flex flex-col pb-20">
      <div className="sticky top-0 z-50 w-full bg-black aspect-video max-h-[45vh] sm:max-h-[55vh] flex items-center justify-center relative overflow-hidden shadow-lg">
        {mediaUrl ? (
          <video
            ref={videoRef}
            src={mediaUrl}
            className="w-full h-full object-contain"
            controls
            autoPlay
            playsInline
          />
        ) : (
          <div className="text-gray-500 text-sm">No media URL found</div>
        )}

        <button
          onClick={() => window.history.length > 1 ? window.history.back() : navigate({ to: '/' })}
          className="absolute top-3 left-3 z-50 p-2 rounded-full bg-black/60 text-white backdrop-blur-md hover:bg-black/80 transition-all"
        >
          <ArrowLeft className="w-5 h-5"/>
        </button>
      </div>

      <div className="flex-1 w-full max-w-4xl mx-auto px-4 py-4 space-y-4">
        <div>
          <h1 className="text-lg sm:text-xl font-bold text-white line-clamp-2">
            {video.title || video.caption || 'Untitled Video'}
          </h1>
          <p className="text-xs text-gray-400 mt-1">
            {video.views_count ? `${video.views_count} views • ` : ''}{timeAgo}
          </p>
        </div>

        <div className="flex items-center justify-between py-3 border-y border-white/10">
          <div className="flex items-center gap-3">
            <Avatar className="w-10 h-10 border border-white/10">
              <AvatarImage ''} src="{video.user?.avatar_url" ||/>
              <AvatarFallback className="bg-pink-600 text-white font-bold">
                {video.user?.username?.charAt(0)?.toUpperCase() || 'U'}
              </AvatarFallback>
            </Avatar>
            <div>
              <p className="font-semibold text-sm text-white">
                {video.user?.full_name || video.user?.username || 'Creator'}
              </p>
              <p className="text-xs text-gray-400">@{video.user?.username || 'user'}</p>
            </div>
          </div>

          <div className="flex items-center gap-2">
            <Button className="rounded-full border-white/20 bg-white/5 text-white hover:bg-white/10 text-xs px-3" onClick="{handleShare}" size="sm" variant="outline">
              <Share2 className="w-3.5 h-3.5 mr-1.5"/> Share
            </Button>
          </div>
        </div>

        {video.caption && video.caption !== video.title && (
          <div className="p-3 bg-white/5 rounded-xl text-xs text-gray-300 leading-relaxed">
            {video.caption}
          </div>
        )}

        <div className="pt-2">
          <h3 className="font-semibold text-sm text-white mb-3">
            Comments ({comments.length})
          </h3>

          <div className="flex gap-2 mb-4">
            <Input onChange="{(e)" value="{commentText}"> setCommentText(e.target.value)}
              placeholder="Add a comment..."
              className="bg-white/5 border-white/10 text-white placeholder:text-gray-500 rounded-full text-xs h-10"
              onKeyDown={(e) => {
                if (e.key === 'Enter' && commentText.trim()) {
                  addCommentMutation.mutate(commentText);
                }
              }}
            />
            <Button addCommentMutation.isPending} disabled="{!commentText.trim()" onClick="{()" size="sm" ||> addCommentMutation.mutate(commentText)}
              className="rounded-full bg-pink-600 hover:bg-pink-700 text-white px-4 h-10"
            >
              <Send className="w-4 h-4"/>
            </Button>
          </div>

          <div className="space-y-3">
            {comments.map((c: any) => (
              <div key={c.id} className="flex gap-3 items-start">
                <Avatar className="w-7 h-7 mt-0.5">
                  <AvatarImage ''} src="{c.user?.avatar_url" ||/>
                  <AvatarFallback className="bg-gray-700 text-xs text-white">
                    {c.user?.username?.charAt(0)?.toUpperCase() || 'U'}
                  </AvatarFallback>
                </Avatar>
                <div className="flex-1 text-xs">
                  <span className="font-semibold text-gray-300 mr-2">
                    @{c.user?.username || 'user'}
                  </span>
                  <span className="text-gray-100">{c.content}</span>
                </div>
              </div>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
}
